# vae-embeddings (chemical_vae / pvae) — code/others/vae-embeddings
# Produces images/vae_emb.sif. CPU-only: this is a TensorFlow 1.x model, no CUDA.
#
# This whole stack is frozen in the TF1 / Python 3.7 era and the pins are tight and
# interdependent — do NOT bump them piecemeal:
#   * tensorflow 1.15.0 is the last TF1 release and the newest TF that supports
#     Python 3.7 (the base image's interpreter); TF2 would break the graph/session
#     code in chemical_vae.
#   * numpy 1.18.5 is the highest numpy TF 1.15 accepts (TF 1.15 pins numpy<1.19).
#   * rdkit-pypi is used (not the `rdkit` package) because prebuilt rdkit wheels for
#     Python 3.7 are only published under the rdkit-pypi name.
#   * mordred 1.2.0 needs numpy<2.0 — already satisfied by the 1.18.5 pin above.
#   * protobuf must stay <3.20: TF 1.15 ships _pb2.py files generated by an old protoc.
#     Left unpinned, pip resolves protobuf to whatever grpcio's install pulls in
#     latest (4.x), whose C++-backed upb implementation refuses to load those old
#     generated descriptors ("Descriptors cannot not be created directly").
tensorflow==1.15.0
protobuf==3.19.6
numpy==1.18.5
pandas==1.3.5
scikit-learn==0.24.2
tqdm==4.65.0
